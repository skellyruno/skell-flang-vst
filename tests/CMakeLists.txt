cmake_minimum_required(VERSION 3.18)
project(SkellFlangPortableDSPTests LANGUAGES CXX)
enable_testing()
add_executable(flang-dsp-tests test_dsp.cpp)
target_compile_features(flang-dsp-tests PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flang-dsp-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME flang-dsp-tests COMMAND flang-dsp-tests)

# Run the measured benchmark only in the ordinary Release Linux job.
# Sanitizer builds compile it but skip timing analysis.
add_executable(flang-dsp-bench bench_dsp.cpp)
target_compile_features(flang-dsp-bench PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flang-dsp-bench PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# Additional portable release gates: local-world wire protocol and
# allocation-free stereo sample processing through all 8 internal engines.
add_executable(flang-world-protocol test_world_protocol.cpp)
target_compile_features(flang-world-protocol PRIVATE cxx_std_17)
add_test(NAME flang-world-protocol COMMAND flang-world-protocol)
add_executable(flang-realtime-allocation test_realtime_alloc.cpp)
target_compile_features(flang-realtime-allocation PRIVATE cxx_std_17)
add_test(NAME flang-realtime-allocation COMMAND flang-realtime-allocation)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(t flang-world-protocol flang-realtime-allocation)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endforeach()
endif()

set(SKELLY_FLANGUS_SOURCE_DIR "" CACHE PATH "Private Skelly Flangus DSP package directory")
if(NOT SKELLY_FLANGUS_SOURCE_DIR STREQUAL "")
  foreach(f Flangus.cpp SkellFlangusAdapter.h core/FlangusFastDist.h)
    if(NOT EXISTS "${SKELLY_FLANGUS_SOURCE_DIR}/${f}")
      message(FATAL_ERROR "Missing private Flangus source: ${f}")
    endif()
  endforeach()
  add_executable(flang-raw-path test_raw_flangus_path.cpp "${SKELLY_FLANGUS_SOURCE_DIR}/Flangus.cpp")
  target_include_directories(flang-raw-path PRIVATE ../Source "${SKELLY_FLANGUS_SOURCE_DIR}")
  target_compile_features(flang-raw-path PRIVATE cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(flang-raw-path PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME flang-raw-path COMMAND flang-raw-path)
endif()

add_executable(flang-raw-settings test_raw_settings.cpp)
target_compile_features(flang-raw-settings PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flang-raw-settings PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME flang-raw-settings COMMAND flang-raw-settings)
option(SKELLY_FLANGUS_SAMPLE_CLOCK_RAMP "Use v2 sample-clock raw gain slew" OFF)
if(SKELLY_FLANGUS_SAMPLE_CLOCK_RAMP)
  if(NOT TARGET flang-raw-path)
    message(FATAL_ERROR "Sample-clock ramp requires the private v2 DSP package")
  endif()
  target_compile_definitions(flang-raw-path PRIVATE SKELLFLANG_SAMPLE_CLOCK_RAMP=1)
endif()

if(TARGET flang-raw-path)
  add_executable(flang-raw-path-lifecycle test_raw_path_lifecycle.cpp "${SKELLY_FLANGUS_SOURCE_DIR}/Flangus.cpp")
  target_include_directories(flang-raw-path-lifecycle PRIVATE ../Source "${SKELLY_FLANGUS_SOURCE_DIR}")
  target_compile_features(flang-raw-path-lifecycle PRIVATE cxx_std_17)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(flang-raw-path-lifecycle PRIVATE -Wall -Wextra -Werror)
  endif()
  if(SKELLY_FLANGUS_SAMPLE_CLOCK_RAMP)
    target_compile_definitions(flang-raw-path-lifecycle PRIVATE SKELLFLANG_SAMPLE_CLOCK_RAMP=1)
  endif()
  add_test(NAME flang-raw-path-lifecycle COMMAND flang-raw-path-lifecycle)
endif()
